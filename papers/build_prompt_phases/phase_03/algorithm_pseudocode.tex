\begin{algorithm}[p]
\caption{\textbf{Pseudo-code of the DT-GSK algorithm.} The junior and senior
gaining-sharing \emph{vector-update equations} are retained unchanged; the
generation around them is not---crossover masking, tie handling, the population
size, and the senior partition are all modified. The added subsystems and their
fixed execution order are listed in Section~\ref{sec:algorithm}.}
\label{alg:dt-gsk}
% 1.8 was chosen for an uncrowded, page-balanced look, but at that stretch the
% float overflows the page (N-014) -- which defeats the goal it was set for.
% 1.55 keeps the open feel and fits.
\setstretch{1.55}
\begin{algorithmic}[1]
\Require objective $f$, dimension $D$, bounds $[\ell,u]$, budget $\mathrm{MaxFES}$ (suite-dependent; Section~\ref{sec:alg:complexity}), seed
\Ensure global best solution $x^{gb}$
\Statex
\Statex \emph{Initialization}
\State Initialize the interaction graphs $G_{\mathrm{abs}}, G_{\mathrm{signed}} \gets 0$ (allocated only at $D\ge50$)
\State Set $NP{=}5D$ and the GSK parameters $KF,\ KR,\ K_{\exp},\ p$
\State Open the 13 RNG substreams for dimension $D$
\State Create a random population $\{x_i\}_{i=1}^{NP}$ in $[\ell,u]$
\State Evaluate $f(x_i)$ for all $i$ and set the global best $x^{gb}$
\Statex
\While{$t < \mathrm{MaxFES}$}
  \Statex
  \Statex \quad\emph{Adapt and build trials}
  \State Reduce $NP$ by the NLPSR schedule \Comment{Eq.~(\ref{eq:nlpsr})}
  \State Assign each individual its $(KF, KR, K_{\exp})$ by the ACE selector
  \State Count the gained and shared dimensions \Comment{Eq.~(\ref{eq:kexp-schedule})}
  \State Apply the junior gaining-sharing phase
  \State Apply the senior gaining-sharing phase
  \State Form the crossover mask: a tier-dependent fraction of rows take a linkage block (learned at $D{\ge}50$, random otherwise, and only when the confidence gate passes); the remaining rows take the $KR$ mask \Comment{Eq.~(\ref{eq:kr-mask})}
  \State Repair the out-of-bound coordinates
  \State Evaluate the maximal prefix of $\{v_i\}$ that fits the remaining budget \Comment{unevaluated rows are non-selectable}
  \State Greedily keep each no-worse trial and update the diversity archive \Comment{Eq.~(\ref{eq:greedy-accept})}
  \Statex
  \Statex \quad\emph{Learn, escape, and refine \textnormal{(dimension-gated)}}
  \State \textbf{if} $D{\ge}50$: update the interaction-structure memory \Comment{Eq.~(\ref{eq:sgsm-graph})}
  \State Update the ACE credit and prune stalled arms (ARGP)
  \State \textbf{if} stagnating: apply the budget-safe escape \Comment{Eq.~(\ref{eq:bse-cauchy})}
  \State Run the charged coordinate local search on the elite rows
  \Statex \quad\emph{Note.}~\textnormal{The $D{\ge}100$ upper-tier controllers are cross-cutting, not a stage after the local search: they act at population reduction (subspace-sampling floor), trial construction (frozen-streak broaden and late linkage random-mix), acceptance (late-acceptance clip and basin-memory update), and escape (basin-novelty reseed); a budget/ROI gate additionally gates the local search (a matching hook exists on the escape but never blocks it in the frozen configuration).}
  \State \textbf{if} $D{\ge}50$: polish the working incumbent $x^{*}$ once, in the final budget slice \Comment{Eq.~(\ref{eq:eigen-polish})}
  \Statex
  \Statex \quad\emph{Update and restart}
  \State Update the global best $x^{gb}$
  \State \textbf{if} deep-stalled: restart the population, keeping $x^{gb}$
\EndWhile
\Statex
\State \Return $x^{gb}$
\end{algorithmic}
\end{algorithm}
